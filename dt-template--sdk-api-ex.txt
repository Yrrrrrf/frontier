# @sdk/api

Enterprise-grade TypeScript HTTP & API SDK built on **Axios 1.20+**, providing
resilient networking, concurrency-safe session management, type-safe CRUD
generation, schema validation, and TanStack React Query integration.

---

## Features

- **Owned Axios Engine**: Every client instance owns its isolated
  `AxiosInstance` with full access to native interceptors, adapters,
  serializers, and transforms via `client.axios`.
- **Race-Safe Authentication**: Single-flight refresh token mutex
  (`refreshToken`) prevents thundering herds on concurrent `401`s. Session
  revision tracking (`authVersion`) ensures stale requests or refreshes never
  overwrite new sign-ins or replay under the wrong identity.
- **Security-First URL & Origin Policy**: Strict destination validation guards
  against backslashes, userinfo, and protocol-relative bypasses. Managed
  authorization headers and cookies are strictly constrained to trusted base
  URLs and explicit `allowedOrigins`.
- **Intelligent Resilience & Deadlines**: Exponential backoff with full jitter,
  automated HTTP `Retry-After` parsing (supporting delta seconds and RFC HTTP
  dates), idempotent mutation safety guards (`replayMutations: false` by
  default), and global operation deadlines (`timeoutMs`).
- **Standardized CRUD Factory**: `createCrud()` produces strongly-typed resource
  clients (`list`, `get`, `create`, `update`, `patch`, `delete`) with path
  segment encoding, traversal prevention, and pluggable schema validation.
- **TanStack React Query Adapter**: Subpath export `@sdk/api/react-query` offers
  hooks and query options (`useResourceList`, `useResourceItem`,
  `useResourceMutation`) with coordinated query key derivation, automatic cache
  invalidation, and parent signal linking (`AbortSignal.any`).
- **Flexible Schema Validation**: Structural parser contract
  (`ResponseValidator`) works seamlessly with Zod 3/4, Valibot, ArkType, or
  standard functions without bundling heavy validation libraries into the core
  runtime.
- **Log Sanitization & Telemetry**: `ApiError.prototype.toJSON()` strips
  sensitive query strings, payload data, headers, and native cause references to
  prevent accidental credential leakage in observability logs (Sentry, Datadog).

## Quick Start

### Basic HTTP Requests

```ts
import { http, request } from "@sdk/api";

interface User {
	id: number;
	name: string;
	email: string;
}

// Typed convenience methods
const users = await http.get<User[]>("/api/users");

// Post with body
const newUser = await http.post<User>("/api/users", {
	name: "Grace Hopper",
	email: "grace@navy.mil",
});

// Generic request call
const profile = await request<User>("/api/me", {
	timeoutMs: 5_000,
	retries: 3,
});
```

### Authentication & Token Management

```ts
import { api, getAuthToken, onUnauthorized, setAuthToken } from "@sdk/api";

// 1. Authenticate using the built-in auth endpoint
const loginResult = await api.auth.login({
	user: "fernando",
	pwd: "secretpassword",
});
// Login automatically persists the token to the active token store

// 2. Inspect or manually update tokens
const token = getAuthToken();
setAuthToken("custom-jwt-token");

// 3. Listen for session expiration / unauthorized events
const unsubscribe = onUnauthorized((error) => {
	console.warn("Session expired or unauthorized:", error.status);
	// Clear app state and redirect to login
	setAuthToken(null);
});

// Call unsubscribe() when the listener is no longer needed
```

## Client Configuration (`createApiClient`)

Create isolated client instances for microservices, SSR requests, or specialized
APIs.

```ts
import {
	createApiClient,
	createBrowserTokenStore,
	createMemoryTokenStore,
} from "@sdk/api";

const client = createApiClient({
	// Base URL (retains subpath prefixes such as /v1)
	baseURL: "https://api.example.com/v1",

	// Explicit origins permitted to receive managed Bearer tokens
	allowedOrigins: ["https://api.example.com"],

	// Total operation deadline (covering hooks, retries, refresh waits, and body reads)
	timeoutMs: 15_000,

	// Retry policy
	retries: 2,
	retryDelayMs: 200,
	maxRetryDelayMs: 5_000,
	retryJitter: true,
	replayMutations: false, // Guard: only safe methods (GET, HEAD, OPTIONS) replay by default

	// Token storage (use memory for SSR, browser for client-side singletons)
	tokenStore:
		typeof window !== "undefined"
			? createBrowserTokenStore("my_app_auth_token")
			: createMemoryTokenStore(),

	// Automated token refresh handler (single-flight mutex across concurrent 401s)
	async refreshToken({ token, signal }) {
		const res = await fetch("https://api.example.com/v1/auth/refresh", {
			method: "POST",
			headers: { "Content-Type": "application/json" },
			body: JSON.stringify({ refreshToken: token }),
			signal,
		});
		if (!res.ok) return null;
		const data = await res.json();
		return data.accessToken;
	},

	// Lifecycle hooks
	beforeRequest(context) {
		context.headers.set("X-Correlation-ID", crypto.randomUUID());
	},
	afterResponse(context) {
		console.log(
			`[${context.method}] ${context.url} -> ${context.response.status}`,
		);
	},
	onError(error) {
		console.error(`Request failed: ${error.code} (${error.status})`);
	},
});
```

### SSR Guidance

For Server-Side Rendering (SSR), **never** store user tokens on the default
client singleton. Instead, instantiate a per-request client:

```ts
export function createServerClient(sessionToken: string) {
	return createApiClient({
		baseURL: process.env.INTERNAL_API_URL,
		tokenStore: createMemoryTokenStore(sessionToken),
		timeoutMs: 8_000,
	});
}
```

## Native Axios Escape Hatch

Every `ApiClient` exposes its underlying native Axios instance via
`client.axios`. Native calls bypass SDK retries, refresh routines, and
validation while inheriting base configurations and native interceptors:

```ts
import { AxiosHeaders } from "@sdk/api";

// Use native interceptors
const interceptorId = client.axios.interceptors.request.use((config) => {
	config.headers.set("X-Custom-Trace", "123");
	return config;
});

// Drop down to native Axios execution
const nativeResponse = await client.axios.get("/raw-endpoint");
console.log(nativeResponse.data, nativeResponse.headers);

// Eject when complete
client.axios.interceptors.request.eject(interceptorId);
```

## CRUD Resources (`createCrud`)

Generate standardized, type-safe CRUD operations for any entity endpoint:

```ts
import { createCrud } from "@sdk/api";
import type { PaginatedResponse } from "@sdk/api";
import { z } from "zod";

// 1. Define models and schemas
const UserSchema = z.object({
	id: z.number(),
	name: z.string(),
	email: z.string().email(),
	role: z.enum(["admin", "user"]),
});

type User = z.infer<typeof UserSchema>;
type CreateUserDto = Omit<User, "id">;
type UpdateUserDto = Partial<CreateUserDto>;

// 2. Create the CRUD resource
export const usersResource = createCrud<
	User,
	CreateUserDto,
	UpdateUserDto,
	"name" | "email",
	PaginatedResponse<User>
>("/api/users", {
	validateEntity: UserSchema,
});

// 3. Perform operations
// GET /api/users?page=1&perPage=20&sort=name&order=asc
const list = await usersResource.list({
	page: 1,
	perPage: 20,
	sort: "name",
	order: "asc",
});

// GET /api/users/42
const user = await usersResource.get(42);

// POST /api/users
const created = await usersResource.create({
	name: "Ada Lovelace",
	email: "ada@analytical.engine",
	role: "admin",
});

// PUT /api/users/42
const updated = await usersResource.update(42, {
	name: "Augusta Ada King",
});

// PATCH /api/users/42
const patched = await usersResource.patch(42, {
	role: "user",
});

// DELETE /api/users/42
await usersResource.delete(42);
```

## TanStack React Query (`@sdk/api/react-query`)

The `@sdk/api/react-query` entrypoint connects CRUD resources directly to
TanStack React Query v5:

```tsx
import {
	useResourceItem,
	useResourceList,
	useResourceMutation,
} from "@sdk/api/react-query";
import { usersResource } from "./resources/users";

export function UsersList() {
	// Query with reactive list filtering
	const { data, isLoading, error } = useResourceList(usersResource, {
		page: 1,
		perPage: 10,
		sort: "name",
	});

	// Mutation with automated cache invalidation
	const createMutation = useResourceMutation(
		usersResource,
		(
			resource,
			newUser: { name: string; email: string; role: "admin" | "user" },
		) => resource.create(newUser),
	);

	if (isLoading) return <div>Loading users...</div>;
	if (error) return <div>Error loading users: {error.message}</div>;

	return (
		<div>
			<ul>
				{data?.data.map((user) => (
					<li key={user.id}>
						{user.name} ({user.email})
					</li>
				))}
			</ul>

			<button
				onClick={() =>
					createMutation.mutate({
						name: "Alan Turing",
						email: "alan@bletchley.org",
						role: "admin",
					})
				}
			>
				Add User
			</button>
		</div>
	);
}

export function UserDetail({ id }: { id: number }) {
	const { data: user } = useResourceItem(
		usersResource,
		id,
		[usersResource.basePath],
		{
			select: (u) => u.name, // Select partial data
		},
	);

	return <h1>{user}</h1>;
}
```

## Error Handling & Diagnostics

All SDK operations throw an `ApiError` instance with standardized error
classification:

```ts
import { ApiError, getAxiosError, isApiError } from "@sdk/api";

try {
	await client.http.get("/api/protected");
} catch (error) {
	if (isApiError(error)) {
		console.error("SDK Error Code:", error.code);
		console.error("HTTP Status:", error.status); // e.g. 404, 500 (0 for network/abort/timeout)
		console.error("Status Text:", error.statusText);
		console.error("Response Data:", error.data); // Decoded server response body
		console.error("Attempt Count:", error.attempt);

		// Retrieve underlying AxiosError if available
		const axiosError = getAxiosError(error);
		if (axiosError) {
			console.error("Native Axios Code:", axiosError.code);
		}
	}
}
```

### Error Codes

| Code                 | Cause                                                                        |
| -------------------- | ---------------------------------------------------------------------------- |
| `HTTP_ERROR`         | Non-2xx response returned by the server.                                     |
| `NETWORK_ERROR`      | Network partition, DNS resolution failure (`ENOTFOUND`), or connection drop. |
| `TIMEOUT`            | Operation exceeded client `timeoutMs` or adapter timeout.                    |
| `ABORTED`            | Request canceled via `AbortSignal` or client `.dispose()`.                   |
| `VALIDATION_ERROR`   | Response body failed schema validation.                                      |
| `PARSE_ERROR`        | Server returned unparseable JSON for a JSON content type.                    |
| `RESPONSE_TOO_LARGE` | Response body exceeded `maxResponseBytes` / `maxContentLength`.              |
| `REQUEST_TOO_LARGE`  | Upload body exceeded `maxBodyLength`.                                        |
| `CONFIG_ERROR`       | Invalid configuration, malicious URL path, or untrusted destination.         |

### Telemetry Safety

Calling `error.toJSON()` produces a sanitized diagnostics object:

```json
{
	"name": "ApiError",
	"code": "HTTP_ERROR",
	"status": 401,
	"method": "GET",
	"attempt": 1,
	"axiosCode": "ERR_BAD_REQUEST"
}
```

> **Security Note**: Full URLs (which may contain sensitive path parameters or
> query tokens), request/response payloads, and authentication headers are
> omitted from `toJSON()` to prevent accidental secret leakage in centralized
> logging systems.

## Generic session controller

Import `createSessionController` from `@sdk/api/session`; it is an optional pure
entry with no dependency on a specific store package, React or React Query.
Supply your API client and a structural store with `getSnapshot()` / `setState()`.
`createSessionStore<TIdentity>()` from `@sdk/state` is the standard sink.

The controller exposes `login(credentials, requestOptions?)`, `restore(options?)`,
`logout(options?)`, `setScope(scope)` and `dispose()`. It creates the existing
auth API with `persistToken: false`, then commits validated login credentials and
identity only after the active-operation/session guard and cache cleanup succeed.
Auth paths, validators and optional demo policy are passed through `auth`.

Provide `getNamespace(identity, scope)` using application user/scope IDs; never
use a raw token. `beforeSessionChange({ previous, next, signal })` runs when the
namespace changes and is awaited before authenticated reads are enabled. Cancel
old queries and clear sensitive cache data there. The controller publishes
pending state immediately to disable authenticated reads. Same-namespace
restore/token refresh does not churn the cache. Without a namespace callback,
login receives a unique operation namespace; `setScope` requires the callback.
A scope change during an initial pending login/restore cancels that operation
and returns the UI to anonymous; explicitly login/restore in the new scope.

Logout clears local credentials and identity immediately, including pending
login/restore operations. Server or cleanup failure rejects the logout promise
without restoring local identity. A stale operation rejects with AbortError;
handle every returned promise. `dispose()` aborts owned operations and removes
the unauthorized listener; it does not dispose the caller-owned client or store.
For cancellation/disposal followed by UI reuse, mount a fresh controller and
restore explicitly. All authentication-changing calls for that store should go
through the controller. Direct external token-store changes require an explicit
restore/logout transition; they are not a second reactive identity source.

`onUnauthorized` retains one-argument callback compatibility and additionally
passes `{ authVersion }`, captured before observers are invoked. The controller
ignores an obsolete unauthorized event even if an earlier observer started a
new session. Automatic cleanup errors can be observed via `onLifecycleError`.
Transient restore errors preserve the stored credential for an explicit retry,
but expose no authenticated identity and clean up the former namespace.

Run `npm run validate --workspace @sdk/api` for this package's strict types and
behavior tests, or `npm run readiness` at the root for the composition checks.
See `sdk/UPGRADE.md` for a complete recipe and `sdk/UPGRADE-VALIDATION.md` for
fresh evidence and export/environment limitations.

import type { TokenStore } from "./types.ts";

/** Compatibility value; demo tokens are never forwarded to a backend. */
export const DEMO_TOKEN = "demo-admin-token-jwt-secret";

export function isValidAuthToken(token: unknown): token is string {
	return (
		typeof token === "string" && token.length > 0 && !/[^\x21-\x7e]/.test(token)
	);
}

export function createMemoryTokenStore(
	initialToken: string | null = null,
): TokenStore {
	let token = initialToken;
	return {
		get: () => token,
		set: (value) => {
			token = value;
		},
	};
}

/** Lazy browser access: denied storage falls back to memory; reads observe other tabs. */
export function createBrowserTokenStore(
	key = "auth_token",
	storage?: Pick<Storage, "getItem" | "setItem" | "removeItem">,
): TokenStore {
	let memory: string | null = null;
	let failed = false;
	const target = () =>
		storage ??
		(typeof window === "undefined" ? undefined : window.localStorage);
	return {
		get() {
			if (!failed) {
				try {
					const current = target();
					if (current) memory = current.getItem(key);
				} catch {
					failed = true;
				}
			}
			return memory;
		},
		set(token) {
			memory = token;
			if (!failed) {
				try {
					const current = target();
					if (token === null) current?.removeItem(key);
					else current?.setItem(key, token);
				} catch {
					failed = true;
				}
			}
		},
	};
}

interface ViteEnv {
	DEV?: boolean;
	PROD?: boolean;
	MODE?: string;
	VITE_ENABLE_DEMO_MODE?: string;
	VITE_API_ORIGIN?: string;
	VITE_BASE_API_URL?: string;
}

function environments() {
	const vite = (import.meta as ImportMeta & { env?: ViteEnv }).env;
	const node = (
		globalThis as typeof globalThis & {
			process?: { env?: Record<string, string | undefined> };
		}
	).process?.env;
	return { vite, node };
}

/** A production environment always wins over development/demo flags. */
export function isDemoAllowed(): boolean {
	const { vite, node } = environments();
	if (
		vite?.PROD ||
		vite?.MODE === "production" ||
		node?.NODE_ENV === "production"
	) {
		return false;
	}
	return (
		vite?.DEV === true ||
		vite?.VITE_ENABLE_DEMO_MODE === "true" ||
		node?.NODE_ENV === "test" ||
		node?.VITE_ENABLE_DEMO_MODE === "true"
	);
}

export function getDefaultBaseUrl(): string {
	const { vite, node } = environments();
	const configured =
		vite?.VITE_API_ORIGIN ||
		vite?.VITE_BASE_API_URL ||
		node?.VITE_API_ORIGIN ||
		node?.VITE_BASE_API_URL;
	// Compatibility: legacy environment settings describe an origin; built-in routes include /api.
	// Explicit createApiClient({ baseUrl }) retains the complete prefix.
	return configured ? configured.replace(/\/+$/, "").replace(/\/api$/, "") : "";
}

/** Own one Axios instance and one auth/refresh lifecycle per SSR request or browser session. */
export function createApiClient(
	configuration: ApiClientOptions = {},
): ApiClient {
	const options = { ...configuration };
	if (
		options.baseUrl !== undefined &&
		options.baseURL !== undefined &&
		options.baseUrl !== options.baseURL
	) {
		throw configError("Use baseUrl or baseURL, not conflicting values");
	}
	const baseUrl = validateBase(
		options.baseURL ?? options.baseUrl ?? getDefaultBaseUrl(),
	);
	const allowed = validateOrigins(options.allowedOrigins ?? []);
	const timeoutMs = nonnegative(options.timeoutMs ?? 15_000, "timeoutMs");
	const defaultLimit = nonnegative(
		options.maxResponseBytes ?? 10 * 1024 * 1024,
		"maxResponseBytes",
	);
	const retries = retryCount(options.retries ?? 2);
	const retryDelayMs = nonnegative(options.retryDelayMs ?? 200, "retryDelayMs");
	const maxRetryDelayMs = nonnegative(
		options.maxRetryDelayMs ?? 5_000,
		"maxRetryDelayMs",
	);
	const refreshTimeoutMs = nonnegative(
		options.refreshTimeoutMs ?? 10_000,
		"refreshTimeoutMs",
	);
	const store = options.tokenStore ?? createMemoryTokenStore();
	const owner = new AbortController();
	const listeners = new Set<
		(
			error: ApiError,
			context: { readonly authVersion: number },
		) => Awaitable<void>
	>();
	let revision = 0;
	let sessionRevision = 0;
	let observedToken: string | null = null;
	let refreshing: Promise<string | null> | undefined;

	const getAuthToken = () => {
		let current: string | null;
		try {
			current = store.get();
		} catch (cause) {
			throw configError("Token store read failed", cause);
		}
		if (current !== null && !isValidAuthToken(current)) {
			throw configError("Token store returned an invalid token");
		}
		if (observedToken !== current) {
			observedToken = current;
			revision++;
			sessionRevision++;
		}
		return current;
	};
	const writeToken = (token: string | null, fromRefresh = false) => {
		if (token !== null && !isValidAuthToken(token)) {
			throw configError("Invalid auth token");
		}
		try {
			store.set(token);
		} catch (cause) {
			throw configError("Token store write failed", cause);
		}
		observedToken = token;
		revision++;
		if (!fromRefresh) sessionRevision++;
	};
	const setAuthToken = (token: string | null) => writeToken(token);
	const notifyUnauthorized = (error: ApiError) => {
		const context = Object.freeze({ authVersion: sessionRevision });
		for (const listener of listeners) {
			try {
				void Promise.resolve(listener(error, context)).catch(() => {});
			} catch {
				/* Observer only. */
			}
		}
	};

	const {
		baseUrl: _base,
		fetch: injectedFetch,
		headers: defaultHeaders,
		credentials,
		timeoutMs: _timeout,
		maxResponseBytes: _limit,
		tokenStore: _store,
		allowedOrigins: _origins,
		retries: _retries,
		retryDelayMs: _delay,
		maxRetryDelayMs: _maxDelay,
		retryJitter: _jitter,
		replayMutations: _replay,
		beforeRequest: _before,
		afterResponse: _after,
		onError: _error,
		refreshToken: _refresh,
		refreshTimeoutMs: _refreshTimeout,
		...nativeDefaults
	} = options;
	const nativeRequestTransforms = Array.isArray(axios.defaults.transformRequest)
		? axios.defaults.transformRequest
		: axios.defaults.transformRequest
			? [axios.defaults.transformRequest]
			: [];
	// Axios supports null constructors at runtime; its declarations omit null.
	const fetchEnvironment = injectedFetch
		? ({
				...options.env,
				fetch: injectedFetch,
				Request: null,
				Response: globalThis.Response,
			} as unknown as NonNullable<AxiosRequestConfig["env"]>)
		: undefined;
	const instance = axios.create({
		...nativeDefaults,
		baseURL: baseUrl,
		headers: toAxiosHeaders(defaultHeaders),
		timeout: options.timeout ?? 0,
		maxRedirects: options.maxRedirects ?? 0,
		maxContentLength:
			options.maxContentLength ?? (defaultLimit === 0 ? -1 : defaultLimit),
		maxBodyLength: options.maxBodyLength ?? 10 * 1024 * 1024,
		paramsSerializer: options.paramsSerializer ?? { indexes: null },
		transformRequest: options.transformRequest ?? [
			prepareBody,
			...nativeRequestTransforms,
		],
		transformResponse: options.transformResponse ?? [parseResponse],
		transitional: {
			clarifyTimeoutError: true,
			forcedJSONParsing: false,
			validateStatusUndefinedResolves: false,
			...options.transitional,
		},
		...(options.adapter !== undefined
			? { adapter: options.adapter }
			: injectedFetch
				? { adapter: "fetch" }
				: {}),
		...(fetchEnvironment ? { env: fetchEnvironment } : {}),
		...(credentials !== undefined && options.withCredentials === undefined
			? { withCredentials: credentials === "include" }
			: {}),
		fetchOptions: { redirect: "error", ...options.fetchOptions },
	});

	// Registered once. Default Axios LIFO ordering puts subsequently added request hooks before this guard.
	const authInterceptor = instance.interceptors.request.use(
		async (rawConfig) => {
			const config = rawConfig as InternalConfig;
			const state = config[attemptState];
			throwIfAborted(owner.signal);
			if (state) throwIfAborted(state.signal);
			try {
				const uri = instance.getUri(config);
				const destination = resolveUrl(baseUrl, uri, allowed);
				const headers = config.headers;
				if (
					!destination.trusted &&
					(headers.has("Authorization") ||
						headers.has("Cookie") ||
						config.auth ||
						config.withCredentials === true)
				) {
					throw configError(
						"Credentials require a trusted baseURL or allowedOrigins",
					);
				}
				const token = state?.skipAuth ? null : getAuthToken();
				if (state) {
					state.sentToken = token;
					state.sentRevision = revision;
					state.sentSession = sessionRevision;
				}
				if (
					!state?.skipAuth &&
					destination.trusted &&
					!config.auth &&
					!headers.has("Authorization") &&
					token &&
					token !== DEMO_TOKEN
				) {
					headers.set("Authorization", `Bearer ${token}`);
					if (state) state.managedAuth = true;
				}
				const context: RequestContext = {
					url: uri,
					method: (config.method ?? "GET").toUpperCase(),
					attempt: state?.attempt ?? 1,
					headers,
					signal: state?.signal ?? owner.signal,
				};
				if (state) state.context = context;
				if (options.beforeRequest) {
					await waitFor(
						Promise.resolve().then(() => options.beforeRequest!(context)),
						context.signal,
					);
				}
				if (
					!destination.trusted &&
					(headers.has("Authorization") || headers.has("Cookie"))
				) {
					throw configError(
						"Header hook added credentials for an untrusted origin",
					);
				}
				if (state) {
					state.managedAuth =
						state.managedAuth &&
						headers.get("Authorization") === `Bearer ${state.sentToken}`;
				}
				return config;
			} catch (cause) {
				if (state) throwIfAborted(state.signal);
				throw isApiError(cause)
					? cause
					: configError("Axios request interceptor failed", cause);
			}
		},
	);

	const refresh = (): Promise<string | null> => {
		if (refreshing) return refreshing;
		const token = getAuthToken();
		const atRevision = revision;
		const scope = createScope(undefined, owner.signal, refreshTimeoutMs);
		const promise = waitFor(
			Promise.resolve().then(() =>
				options.refreshToken!({ token, signal: scope.signal }),
			),
			scope.signal,
		)
			.then((nextToken) => {
				throwIfAborted(scope.signal);
				const current = getAuthToken();
				if (revision !== atRevision) return current;
				if (nextToken !== null) writeToken(nextToken, true);
				return nextToken;
			})
			.finally(() => {
				scope.close();
				if (refreshing === promise) refreshing = undefined;
			});
		refreshing = promise;
		return promise;
	};

	async function execute<T, B = unknown, P extends object = QueryParams>(
		path: string,
		input: RequestOptions<B, P, T> = {},
	): Promise<AxiosResponse<unknown, B, {}, P>> {
		const budget = nonnegative(input.timeoutMs ?? timeoutMs, "timeoutMs");
		const scope = createScope(input.signal, owner.signal, budget);
		const startedAt = Date.now();
		let url = "";
		let method = "GET";
		let attempt = 0;
		try {
			throwIfAborted(scope.signal);
			const {
				body,
				data,
				bodyFactory,
				bodyEncoding = "auto",
				responseType = "auto",
				validate,
				skipAuth = false,
				skipRefresh = false,
				method: requestedMethod,
				headers,
				signal: _signal,
				timeoutMs: _budget,
				retries: _retries,
				retryDelayMs: _delay,
				maxRetryDelayMs: _maxDelay,
				retryJitter: _jitter,
				replayMutations: _replay,
				maxResponseBytes,
				credentials,
				redirect,
				...nativeConfig
			} = input;
			method = (requestedMethod ?? "GET").toUpperCase();
			if (!/^[!#$%&'*+.^_`|~\da-z-]+$/i.test(method)) {
				throw configError("Invalid HTTP method");
			}
			if (
				(body !== undefined && data !== undefined) ||
				(bodyFactory && (body !== undefined || data !== undefined))
			) {
				throw configError("Use exactly one of body, data or bodyFactory");
			}
			if (
				(method === "GET" || method === "HEAD") &&
				(body !== undefined || data !== undefined || bodyFactory)
			) {
				throw configError("GET and HEAD cannot have a body");
			}
			if (
				![
					"auto",
					"json",
					"text",
					"blob",
					"arraybuffer",
					"arrayBuffer",
					"stream",
					"document",
					"formdata",
				].includes(responseType)
			) {
				throw configError("Invalid responseType");
			}
			if (!["auto", "json", "raw"].includes(bodyEncoding)) {
				throw configError("Invalid bodyEncoding");
			}
			url = resolveUrl(baseUrl, path, allowed).url;
			const count = retryCount(input.retries ?? retries);
			const delay = nonnegative(
				input.retryDelayMs ?? retryDelayMs,
				"retryDelayMs",
			);
			const maxDelay = nonnegative(
				input.maxRetryDelayMs ?? maxRetryDelayMs,
				"maxRetryDelayMs",
			);
			const payload = body !== undefined ? body : data;
			const safeToReplay =
				(!isStreamBody(payload) || Boolean(bodyFactory)) &&
				(SAFE_METHODS.has(method) ||
					(input.replayMutations ?? options.replayMutations ?? false));
			const statusPolicy =
				input.validateStatus ??
				options.validateStatus ??
				((status) => status >= 200 && status < 300);
			const validateStatus = (status: number) => {
				const accepted = statusPolicy(status);
				if (typeof accepted !== "boolean") {
					throw configError("validateStatus must return a boolean");
				}
				return accepted;
			};
			let retried = 0;
			let authReplayed = false;
			for (;;) {
				throwIfAborted(scope.signal);
				attempt++;
				const state = Object.assign(new AttemptState(), {
					attempt,
					signal: scope.signal,
					skipAuth,
					sentRevision: revision,
					sentSession: sessionRevision,
				});
				let response: AxiosResponse<unknown, B, {}, P> | undefined;
				let afterRan = false;
				const after = async (value: AxiosResponse<unknown>) => {
					afterRan = true;
					if (options.afterResponse && state.context) {
						try {
							await waitFor(
								Promise.resolve().then(() =>
									options.afterResponse!({
										...state.context!,
										response: value,
									}),
								),
								scope.signal,
							);
						} catch (cause) {
							throwIfAborted(scope.signal);
							throw configError("afterResponse failed", cause);
						}
					}
				};
				try {
					let outgoing = payload;
					if (bodyFactory) {
						try {
							outgoing = bodyFactory();
						} catch (cause) {
							throw configError("bodyFactory failed", cause);
						}
					}
					const requestHeaders = mergeHeaders(defaultHeaders, headers);
					if (bodyEncoding === "json") {
						requestHeaders.setContentType("application/json", true);
						if (typeof outgoing === "string") {
							outgoing = JSON.stringify(outgoing) as B;
						}
					}
					if (
						bodyEncoding === "raw" &&
						outgoing !== undefined &&
						!isStreamBody(outgoing) &&
						typeof outgoing === "object" &&
						outgoing !== null &&
						!ArrayBuffer.isView(outgoing) &&
						!(outgoing instanceof ArrayBuffer) &&
						!(typeof Blob !== "undefined" && outgoing instanceof Blob) &&
						!(
							typeof FormData !== "undefined" && outgoing instanceof FormData
						) &&
						!(outgoing instanceof URLSearchParams)
					) {
						throw configError(
							"raw body requires a transport-supported representation",
						);
					}
					const config: AxiosRequestConfig<B, P> & {
						[attemptState]: AttemptState;
						[responseMode]: string;
					} = {
						...nativeConfig,
						url,
						baseURL: "",
						method,
						headers: requestHeaders,
						validateStatus,
						signal:
							responseType === "stream"
								? AbortSignal.any([
										scope.signal,
										input.signal ?? scope.signal,
										owner.signal,
									])
								: scope.signal,
						responseType:
							responseType === "auto"
								? "text"
								: responseType === "arrayBuffer"
									? "arraybuffer"
									: responseType,
						[responseMode]: responseType,
						[attemptState]: state,
						...(outgoing === undefined ? {} : { data: outgoing }),
						...(maxResponseBytes === undefined
							? {}
							: {
									maxContentLength:
										nonnegative(maxResponseBytes, "maxResponseBytes") === 0
											? -1
											: maxResponseBytes,
								}),
						...(credentials !== undefined && input.withCredentials === undefined
							? { withCredentials: credentials === "include" }
							: {}),
						fetchOptions: {
							...instance.defaults.fetchOptions,
							...input.fetchOptions,
							...(redirect ? { redirect } : {}),
						},
					};
					if (bodyEncoding === "raw" && !input.transformRequest) {
						config.transformRequest = [prepareBody, (value) => value];
					}
					const pending = instance.request<
						unknown,
						AxiosResponse<unknown, B, {}, P>,
						B,
						P
					>(config);
					void pending.then(
						(late) => {
							if (scope.signal.aborted) discardData(late.data);
						},
						() => {},
					);
					response = await waitFor(pending, scope.signal);
					throwIfAborted(scope.signal);
					await after(response);
					if (
						responseType === "arrayBuffer" &&
						ArrayBuffer.isView(response.data)
					) {
						const view = response.data;
						response.data = new Uint8Array(
							view.buffer,
							view.byteOffset,
							view.byteLength,
						).slice().buffer;
					}
					if (validate) {
						try {
							response.data = await waitFor(
								validateResponse(validate, response.data),
								scope.signal,
							);
						} catch (cause) {
							throwIfAborted(scope.signal);
							throw new ApiError(
								response.status,
								"Response validation failed",
								response.data,
								url,
								{
									code: "VALIDATION_ERROR",
									method,
									attempt,
									headers: response.headers,
									cause,
								},
							);
						}
					}
					return response;
				} catch (cause) {
					if (response) discardData(response.data);
					throwIfAborted(scope.signal);
					let finalCause: unknown = cause;
					if (!afterRan && isAxiosError(cause) && cause.response) {
						try {
							await after(cause.response);
						} catch (hookError) {
							finalCause = hookError;
						}
					}
					const error = fromAxiosError(
						finalCause,
						state.context?.url ?? url,
						method,
						attempt,
					);
					discardData(error.data);
					if (error.status === 401 && !skipAuth) {
						const current = getAuthToken();
						const unchanged =
							revision === state.sentRevision && current === state.sentToken;
						if (
							!skipRefresh &&
							!authReplayed &&
							safeToReplay &&
							state.managedAuth &&
							options.refreshToken &&
							state.sentSession === sessionRevision
						) {
							authReplayed = true;
							let token: string | null = current;
							if (unchanged) {
								try {
									token = await waitFor(refresh(), scope.signal);
								} catch {
									throwIfAborted(scope.signal);
									token = null;
								}
							}
							if (
								token &&
								token !== DEMO_TOKEN &&
								state.sentSession === sessionRevision
							) {
								continue;
							}
						}
						if (
							getAuthToken() === state.sentToken &&
							revision === state.sentRevision &&
							state.sentSession === sessionRevision
						) {
							notifyUnauthorized(error);
						}
						throw error;
					}
					if (!safeToReplay || !isTransient(error) || retried >= count) {
						throw error;
					}
					const serverDelay = retryAfter(
						String(error.headers.get("Retry-After") ?? "") || null,
					);
					const exponential = Math.min(maxDelay, delay * 2 ** retried);
					const wait =
						serverDelay ??
						((input.retryJitter ?? options.retryJitter ?? true)
							? Math.random() * exponential
							: exponential);
					if (
						wait > maxDelay ||
						(budget > 0 && wait >= budget - (Date.now() - startedAt))
					) {
						throw error;
					}
					retried++;
					await pause(wait, scope.signal);
				}
			}
		} catch (cause) {
			const error = fromAxiosError(cause, url, method, attempt);
			if (options.onError && !scope.signal.aborted) {
				try {
					await waitFor(
						Promise.resolve().then(() => options.onError!(error)),
						scope.signal,
					);
				} catch {
					/* Observer only. */
				}
			}
			throw error;
		} finally {
			scope.close();
		}
	}

	async function request<T, B = unknown, P extends object = QueryParams>(
		path: string,
		input?: RequestOptions<B, P, T>,
	): Promise<T> {
		return (await execute(path, input)).data as T;
	}
	async function requestResponse<
		T,
		B = unknown,
		P extends object = QueryParams,
	>(
		path: string,
		input?: RequestOptions<B, P, T>,
	): Promise<AxiosResponse<T, B, {}, P>> {
		return (await execute(path, input)) as AxiosResponse<T, B, {}, P>;
	}
	const http: HttpClient = {
		get: (path, input) => request(path, { ...input, method: "GET" }),
		head: (path, input) =>
			requestResponse<void>(path, { ...input, method: "HEAD" }),
		options: (path, input) => request(path, { ...input, method: "OPTIONS" }),
		post: (path, body, input) =>
			request(path, {
				...input,
				...(body === undefined ? {} : { body }),
				method: "POST",
			}),
		put: (path, body, input) =>
			request(path, {
				...input,
				...(body === undefined ? {} : { body }),
				method: "PUT",
			}),
		patch: (path, body, input) =>
			request(path, {
				...input,
				...(body === undefined ? {} : { body }),
				method: "PATCH",
			}),
		delete: (path, input) => request(path, { ...input, method: "DELETE" }),
	};
	return {
		request,
		requestResponse,
		axios: instance,
		http,
		getAuthToken,
		setAuthToken,
		get authVersion() {
			getAuthToken();
			return sessionRevision;
		},
		isDemoMode: () => getAuthToken() === DEMO_TOKEN,
		onUnauthorized(callback) {
			listeners.add(callback);
			return () => {
				listeners.delete(callback);
			};
		},
		dispose() {
			if (owner.signal.aborted) return;
			owner.abort();
			listeners.clear();
			instance.interceptors.request.eject(authInterceptor);
			// Leave a lightweight closed-owner guard for exposed native calls.
			instance.interceptors.request.use(() => {
				throwIfAborted(owner.signal);
				throw configError("Client disposed");
			});
		},
	};
}

import { defaultClient } from "./client.ts";
import { configError } from "./internal/control.ts";

import type {
	ApiClient,
	PaginatedResponse,
	PaginationParams,
	QueryParams,
	RequestOptions,
	ResponseValidator,
} from "./types.ts";

export type ResourceId = string | number;
export type ResourceRequestOptions<P extends object = QueryParams> = Omit<
	RequestOptions<never, P>,
	| "body"
	| "data"
	| "bodyFactory"
	| "params"
	| "validate"
	| "method"
	| "responseType"
>;

export interface CrudResource<
	TEntity,
	TCreateDto = Partial<TEntity>,
	TUpdateDto = Partial<TEntity>,
	TSortField extends string = string,
	TList = PaginatedResponse<TEntity> | TEntity[],
	TParams extends PaginationParams<TSortField> = PaginationParams<TSortField>,
> {
	readonly basePath: string;
	list(
		params?: TParams,
		options?: ResourceRequestOptions<TParams>,
	): Promise<TList>;
	get(
		id: ResourceId,
		options?: ResourceRequestOptions<TParams>,
	): Promise<TEntity>;
	create(
		dto: TCreateDto,
		options?: ResourceRequestOptions<TParams>,
	): Promise<TEntity>;
	update(
		id: ResourceId,
		dto: TUpdateDto,
		options?: ResourceRequestOptions<TParams>,
	): Promise<TEntity>;
	patch(
		id: ResourceId,
		dto: Partial<TUpdateDto>,
		options?: ResourceRequestOptions<TParams>,
	): Promise<TEntity>;
	delete(
		id: ResourceId,
		options?: ResourceRequestOptions<TParams>,
	): Promise<void>;
}

export interface CrudOptions<TEntity, TList, P extends object = QueryParams> {
	client?: ApiClient;
	request?: ResourceRequestOptions<P>;
	validateEntity?: ResponseValidator<TEntity>;
	validateList?: ResponseValidator<TList>;
}

function idPath(base: string, id: ResourceId): string {
	if (
		(typeof id !== "string" && typeof id !== "number") ||
		id === "" ||
		(typeof id === "number" && !Number.isFinite(id)) ||
		id === "." ||
		id === ".."
	) {
		throw configError("Invalid resource ID");
	}
	return `${base}/${encodeURIComponent(String(id))}`;
}

/** DTO, sorting, filters, and exact list-envelope types remain application-owned. */
export function createCrud<
	TEntity,
	TCreateDto = Partial<TEntity>,
	TUpdateDto = Partial<TEntity>,
	TSortField extends string = string,
	TList = PaginatedResponse<TEntity> | TEntity[],
	TParams extends PaginationParams<TSortField> = PaginationParams<TSortField>,
>(
	basePath: string,
	options: CrudOptions<TEntity, TList, TParams> = {},
): CrudResource<TEntity, TCreateDto, TUpdateDto, TSortField, TList, TParams> {
	if (!basePath || /[?#]/.test(basePath) || basePath.startsWith("//")) {
		throw configError(
			"Resource path must not contain a query, fragment, or protocol-relative URL",
		);
	}
	const base = (
		/^https?:\/\//i.test(basePath) || basePath.startsWith("/")
			? basePath
			: `/${basePath}`
	).replace(/\/+$/, "");
	if (!base) throw configError("Resource path cannot be the root");
	const client = options.client ?? defaultClient;
	const common = (input?: ResourceRequestOptions<TParams>) => ({
		...options.request,
		...input,
		headers: mergeHeaders(options.request?.headers, input?.headers),
	});
	const entity = (input?: ResourceRequestOptions<TParams>) => ({
		...common(input),
		...(options.validateEntity ? { validate: options.validateEntity } : {}),
	});
	return {
		basePath: base,
		list(params, input) {
			return client.request<TList, never, TParams>(base, {
				method: "GET",
				...common(input),
				...(params === undefined ? {} : { params }),
				...(options.validateList ? { validate: options.validateList } : {}),
			});
		},
		get: (id, input) =>
			client.request<TEntity, never, TParams>(idPath(base, id), {
				...entity(input),
				method: "GET",
			}),
		create: (dto, input) =>
			client.request<TEntity, TCreateDto, TParams>(base, {
				...entity(input),
				body: dto,
				method: "POST",
			}),
		update: (id, dto, input) =>
			client.request<TEntity, TUpdateDto, TParams>(idPath(base, id), {
				...entity(input),
				body: dto,
				method: "PUT",
			}),
		patch: (id, dto, input) =>
			client.request<TEntity, Partial<TUpdateDto>, TParams>(idPath(base, id), {
				...entity(input),
				body: dto,
				method: "PATCH",
			}),
		async delete(id, input) {
			await client.request<unknown, never, TParams>(idPath(base, id), {
				...common(input),
				method: "DELETE",
			});
		},
	};
}

import { DEMO_TOKEN, isDemoAllowed, isValidAuthToken } from "./auth.ts";
import { defaultClient } from "./client.ts";
import { configError, throwIfAborted } from "./internal/control.ts";
import { toAxiosHeaders } from "./internal/headers.ts";
import type { ApiClient, RequestOptions, ResponseValidator } from "./types.ts";

export { isDemoAllowed } from "./auth.ts";

export interface AuthUser {
	wiw: string;
	displayName: string;
	email: string;
	isAdmin: boolean;
}

export interface LoginCredentials {
	user: string;
	pwd: string;
}
export interface LoginResponse<TUser = AuthUser> {
	token: string;
	userData: TUser;
}
export type AuthRequestOptions<T = unknown> = Omit<
	RequestOptions<unknown, never, T>,
	"body" | "data" | "bodyFactory" | "method" | "params" | "paramsSerializer"
>;

export interface AuthApiOptions<TUser> {
	loginPath?: string;
	mePath?: string;
	logoutPath?: string;
	validateLogin?: ResponseValidator<LoginResponse<TUser>>;
	validateUser?: ResponseValidator<TUser>;
	/** Defaults to true; false supports applications that manage login state separately. */
	persistToken?: boolean;
	/** Explicit local demo policy. Still disabled by production environment detection. */
	demo?: { credentials: LoginCredentials; user: TUser };
}

const demoUser: AuthUser = {
	wiw: "ADMIN",
	displayName: "Administrador de Prueba",
	email: "admin.test@corp.internal",
	isAdmin: true,
};

function parseAuthUser(input: unknown): AuthUser {
	if (
		!input ||
		typeof input !== "object" ||
		!("wiw" in input) ||
		typeof input.wiw !== "string" ||
		!("displayName" in input) ||
		typeof input.displayName !== "string" ||
		!("email" in input) ||
		typeof input.email !== "string" ||
		!("isAdmin" in input) ||
		typeof input.isAdmin !== "boolean"
	) {
		throw new TypeError("Invalid AuthUser");
	}
	return {
		wiw: input.wiw,
		displayName: input.displayName,
		email: input.email,
		isAdmin: input.isAdmin,
	};
}

function defaultLoginValidator(input: unknown): LoginResponse {
	if (
		!input ||
		typeof input !== "object" ||
		!("token" in input) ||
		!isValidAuthToken(input.token) ||
		!("userData" in input)
	) {
		throw new TypeError("Invalid LoginResponse");
	}
	return { token: input.token, userData: parseAuthUser(input.userData) };
}

/** Generic auth contracts can be replaced without changing the transport or global endpoint object. */
export function createAuthApi<TUser = AuthUser>(
	client: ApiClient = defaultClient,
	settings: AuthApiOptions<TUser> = {},
) {
	let action = 0;
	const assertToken = (result: LoginResponse<TUser>) => {
		if (!result || !isValidAuthToken(result.token)) {
			throw configError("Login response has an invalid token");
		}
	};
	return {
		async login(
			credentials: LoginCredentials,
			input: AuthRequestOptions<LoginResponse<TUser>> = {},
		): Promise<LoginResponse<TUser>> {
			if (
				!credentials ||
				typeof credentials.user !== "string" ||
				typeof credentials.pwd !== "string"
			) {
				throw configError("Invalid login credentials");
			}
			if (input.signal) throwIfAborted(input.signal);
			const activeAction = ++action;
			const session = client.authVersion;
			const demo = settings.demo;
			let result: LoginResponse<TUser>;
			if (
				demo &&
				isDemoAllowed() &&
				credentials.user.trim().toUpperCase() ===
					demo.credentials.user.trim().toUpperCase() &&
				credentials.pwd === demo.credentials.pwd
			) {
				result = { token: DEMO_TOKEN, userData: demo.user };
			} else {
				result = await client.http.post<LoginResponse<TUser>, LoginCredentials>(
					settings.loginPath ?? "/api/Auth/login",
					credentials,
					{
						...input,
						skipAuth: true,
						skipRefresh: true,
						retries: 0,
						replayMutations: false,
						...(settings.validateLogin
							? { validate: settings.validateLogin }
							: {}),
					},
				);
			}
			assertToken(result);
			if (
				settings.persistToken !== false &&
				activeAction === action &&
				client.authVersion === session
			) {
				client.setAuthToken(result.token);
			}
			return result;
		},
		async me(input: AuthRequestOptions<TUser> = {}): Promise<TUser> {
			if (input.signal) throwIfAborted(input.signal);
			if (settings.demo && isDemoAllowed() && client.isDemoMode()) {
				return settings.demo.user;
			}
			return client.http.get<TUser>(settings.mePath ?? "/api/Auth/me", {
				...input,
				...(settings.validateUser ? { validate: settings.validateUser } : {}),
			});
		},
		async logout(input: AuthRequestOptions<void> = {}): Promise<void> {
			action++;
			const token = client.getAuthToken();
			// Invalidate refresh/login immediately, without waiting for the server to acknowledge logout.
			client.setAuthToken(null);
			if (token !== DEMO_TOKEN) {
				const headers = toAxiosHeaders(input.headers);
				if (token && !headers.has("Authorization")) {
					headers.set("Authorization", `Bearer ${token}`);
				}
				await client.http.post<void>(
					settings.logoutPath ?? "/api/Auth/logout",
					undefined,
					{
						...input,
						headers,
						skipAuth: true,
						skipRefresh: true,
						retries: 0,
						replayMutations: false,
					},
				);
			}
		},
	};
}

/** Original endpoint path and credentials stay compatible; demo bypass is production-gated. */
export const api = {
	auth: createAuthApi(defaultClient, {
		demo: { credentials: { user: "ADMIN", pwd: "admin" }, user: demoUser },
		validateLogin: defaultLoginValidator,
		validateUser: parseAuthUser,
	}),
};

import { AxiosHeaders, isAxiosError, isCancel } from "axios";
import type { AxiosError } from "axios";
import type { HeaderInput } from "./types.ts";
import { toAxiosHeaders } from "./internal/headers.ts";

export type ApiErrorCode =
	| "HTTP_ERROR"
	| "NETWORK_ERROR"
	| "TIMEOUT"
	| "ABORTED"
	| "PARSE_ERROR"
	| "VALIDATION_ERROR"
	| "CONFIG_ERROR"
	| "RESPONSE_TOO_LARGE"
	| "REQUEST_TOO_LARGE";

export interface ApiErrorDetails {
	code?: ApiErrorCode;
	method?: string;
	headers?: HeaderInput;
	attempt?: number;
	cause?: unknown;
	axiosCode?: string;
}

/** Keeps payload/cause available for callers, but excludes them from JSON diagnostics. */
export class ApiError<TErrorPayload = unknown> extends Error {
	readonly isApiError = true;
	readonly status: number;
	readonly statusText: string;
	readonly data: TErrorPayload;
	readonly url: string;
	readonly code: ApiErrorCode;
	readonly method: string;
	readonly headers: AxiosHeaders;
	readonly attempt: number;
	readonly axiosCode: string | undefined;

	constructor(
		status: number,
		statusText: string,
		data: TErrorPayload,
		url = "",
		details: ApiErrorDetails = {},
	) {
		const code = details.code ?? "HTTP_ERROR";
		super(
			`${code}${status ? ` ${status}` : ""}${
				statusText ? ` ${statusText}` : ""
			}`,
			{ cause: details.cause },
		);
		this.name = "ApiError";
		this.status = status;
		this.statusText = statusText;
		this.data = data;
		this.url = url;
		this.code = code;
		this.method = details.method ?? "GET";
		this.headers = toAxiosHeaders(details.headers);
		this.attempt = details.attempt ?? 0;
		this.axiosCode = details.axiosCode;
	}

	toJSON() {
		// URLs can carry secrets even in path segments. Payloads and headers can too.
		return {
			name: this.name,
			code: this.code,
			status: this.status,
			method: this.method,
			attempt: this.attempt,
			axiosCode: this.axiosCode,
		};
	}
}

/** T describes the backend payload; use a schema before trusting its contents. */
export function isApiError<T = unknown>(error: unknown): error is ApiError<T> {
	return (
		typeof error === "object" &&
		error !== null &&
		"isApiError" in error &&
		error.isApiError === true &&
		"code" in error &&
		"status" in error
	);
}

export function isAbortError(error: unknown): boolean {
	return isApiError(error)
		? error.code === "ABORTED"
		: isCancel(error) ||
				(error instanceof Error && error.name === "AbortError");
}

export function fromAxiosError(
	error: unknown,
	url: string,
	method: string,
	attempt: number,
): ApiError {
	if (isApiError(error)) {
		return new ApiError(error.status, error.statusText, error.data, url, {
			code: error.code,
			method,
			attempt,
			headers: error.headers,
			cause: error.cause,
			...(error.axiosCode ? { axiosCode: error.axiosCode } : {}),
		});
	}
	if (!isAxiosError(error)) {
		return new ApiError(0, "Request setup or hook failed", undefined, url, {
			code: "CONFIG_ERROR",
			method,
			attempt,
			cause: error,
		});
	}
	const response = error.response;
	let code: ApiErrorCode;
	if (isCancel(error)) code = "ABORTED";
	else if (error.code === "ETIMEDOUT" || error.code === "ECONNABORTED") {
		code = "TIMEOUT";
	} else if (
		/maxContentLength|response.*(?:size|larger)/i.test(error.message)
	) {
		code = "RESPONSE_TOO_LARGE";
	} else if (/maxBodyLength|Request body larger/i.test(error.message)) {
		code = "REQUEST_TOO_LARGE";
	} else if (
		error.code === "ERR_BAD_RESPONSE" &&
		error.cause instanceof SyntaxError
	) {
		code = "PARSE_ERROR";
	} else if (response) code = "HTTP_ERROR";
	else if (
		[
			"ERR_NETWORK",
			"ECONNREFUSED",
			"ECONNRESET",
			"ENOTFOUND",
			"EAI_AGAIN",
			"EPIPE",
		].includes(error.code ?? "")
	) {
		code = "NETWORK_ERROR";
	} else code = "CONFIG_ERROR";
	return new ApiError(
		response?.status ?? 0,
		response?.statusText ?? "Axios request failed",
		response?.data,
		url,
		{
			code,
			method,
			attempt,
			headers: AxiosHeaders.from(
				response?.headers as import("axios").RawAxiosHeaders | undefined,
			),
			cause: error,
			...(error.code ? { axiosCode: error.code } : {}),
		},
	);
}

/** The original AxiosError remains the cause; backend payloads still require validation. */
export function getAxiosError<T = unknown>(
	error: unknown,
): AxiosError<T> | undefined {
	if (isAxiosError<T>(error)) return error;
	return isApiError(error) && isAxiosError<T>(error.cause)
		? error.cause
		: undefined;
}

import {
	queryOptions,
	useMutation,
	useQuery,
	useQueryClient,
} from "@tanstack/react-query";
import type {
	UseMutationOptions,
	UseQueryOptions,
} from "@tanstack/react-query";
import type {
	CrudResource,
	ResourceId,
	ResourceRequestOptions,
} from "./crud.ts";
import type { ApiError } from "./errors.ts";
import type { PaginatedResponse, PaginationParams } from "./types.ts";

/** Include tenant/user/session in this key when the cache is shared across credentials. */
export type ResourceQueryKey = readonly unknown[];
type QueryOverrides<T, TSelected = T> = Omit<
	UseQueryOptions<T, ApiError, TSelected>,
	"queryKey" | "queryFn"
>;

function requestSignal(
	querySignal: AbortSignal,
	parent?: AbortSignal | null,
): AbortSignal {
	return parent ? AbortSignal.any([querySignal, parent]) : querySignal;
}

export function createResourceQueries<
	E,
	C,
	U,
	S extends string,
	L,
	P extends PaginationParams<S>,
>(
	resource: CrudResource<E, C, U, S, L, P>,
	key: ResourceQueryKey = [resource.basePath],
	requestOptions: ResourceRequestOptions<P> = {},
) {
	const namespace = [...key];
	if (!namespace.length) {
		throw new TypeError("Resource query key must not be empty");
	}
	const keys = {
		all: namespace as ResourceQueryKey,
		lists: () => [...namespace, "list"] as const,
		list: (params?: P) => [...namespace, "list", params ?? {}] as const,
		details: () => [...namespace, "detail"] as const,
		detail: (id: ResourceId | null | undefined) =>
			[...namespace, "detail", id] as const,
	};
	return {
		keys,
		list<Selected = L>(params?: P, overrides?: QueryOverrides<L, Selected>) {
			return queryOptions<L, ApiError, Selected>({
				queryKey: keys.list(params),
				retry: false,
				queryFn: ({ signal }) =>
					resource.list(params, {
						...requestOptions,
						signal: requestSignal(signal, requestOptions.signal),
					}),
				...overrides,
			});
		},
		item<Selected = E>(
			id: ResourceId | null | undefined,
			overrides?: QueryOverrides<E, Selected>,
		) {
			return queryOptions<E, ApiError, Selected>({
				queryKey: keys.detail(id),
				retry: false,
				enabled: id !== null && id !== undefined,
				queryFn: ({ signal }) => {
					if (id === null || id === undefined) {
						throw new TypeError("A resource ID is required");
					}
					return resource.get(id, {
						...requestOptions,
						signal: requestSignal(signal, requestOptions.signal),
					});
				},
				...overrides,
			});
		},
	};
}

export function useResourceList<
	E,
	C = Partial<E>,
	U = Partial<E>,
	S extends string = string,
	L = PaginatedResponse<E> | E[],
	P extends PaginationParams<S> = PaginationParams<S>,
	Selected = L,
>(
	resource: CrudResource<E, C, U, S, L, P>,
	params?: P,
	queryKeyPrefix: string | ResourceQueryKey = [resource.basePath],
	options?: QueryOverrides<L, Selected>,
) {
	const key =
		typeof queryKeyPrefix === "string" ? [queryKeyPrefix] : queryKeyPrefix;
	return useQuery<L, ApiError, Selected>(
		createResourceQueries(resource, key).list<Selected>(params, options),
	);
}

export function useResourceItem<
	E,
	C = Partial<E>,
	U = Partial<E>,
	S extends string = string,
	L = PaginatedResponse<E> | E[],
	P extends PaginationParams<S> = PaginationParams<S>,
	Selected = E,
>(
	resource: CrudResource<E, C, U, S, L, P>,
	id: ResourceId | null | undefined,
	queryKeyPrefix: string | ResourceQueryKey = [resource.basePath],
	options?: QueryOverrides<E, Selected>,
) {
	const key =
		typeof queryKeyPrefix === "string" ? [queryKeyPrefix] : queryKeyPrefix;
	return useQuery<E, ApiError, Selected>(
		createResourceQueries(resource, key).item<Selected>(id, options),
	);
}

/** Successful mutations invalidate the complete resource namespace, including filtered lists. */
export function useResourceMutation<
	E,
	C,
	U,
	S extends string,
	L,
	P extends PaginationParams<S>,
	V,
>(
	resource: CrudResource<E, C, U, S, L, P>,
	mutationFn: (
		resource: CrudResource<E, C, U, S, L, P>,
		variables: V,
	) => Promise<E | void>,
	key: ResourceQueryKey = [resource.basePath],
	options?: Omit<UseMutationOptions<E | void, ApiError, V>, "mutationFn">,
) {
	const queryClient = useQueryClient();
	return useMutation<E | void, ApiError, V>({
		retry: false,
		...options,
		mutationFn: (variables) => mutationFn(resource, variables),
		onSuccess: async (...args) => {
			await queryClient.invalidateQueries({ queryKey: key });
			await options?.onSuccess?.(...args);
		},
	});
}

import {
	createAuthApi,
	type AuthApiOptions,
	type AuthRequestOptions,
	type LoginCredentials,
	type LoginResponse,
} from "./endpoints.ts";
import type { ApiClient, Awaitable } from "./types.ts";

/** Structural sink: no runtime dependency on @sdk/state or React. */
export interface SessionState<TIdentity> {
	readonly identity: TIdentity | null;
	readonly namespace: string | null;
	readonly status:
		| "anonymous"
		| "authenticating"
		| "restoring"
		| "authenticated"
		| "error";
	readonly error: unknown;
}
export interface SessionSink<TIdentity> {
	getSnapshot(): SessionState<TIdentity>;
	setState(state: SessionState<TIdentity>): void;
}
export interface SessionControllerOptions<TIdentity> {
	client: ApiClient;
	store: SessionSink<TIdentity>;
	auth?: Omit<AuthApiOptions<TIdentity>, "persistToken">;
	/** Opaque app user/scope key; never use the raw access token. */
	getNamespace?(identity: TIdentity, scope: string | null): string;
	scope?: string | null;
	/** Awaited before enabling authenticated reads. Cancel queries and clear sensitive caches here. */
	beforeSessionChange?(change: {
		previous: SessionState<TIdentity>;
		next: SessionState<TIdentity>;
		signal: AbortSignal;
	}): Awaitable<void>;
	/** Observer for failures of automatic unauthorized cleanup. */
	onLifecycleError?(error: unknown): void;
}
export function createSessionController<TIdentity>(
	options: SessionControllerOptions<TIdentity>,
) {
	const { client, store } = options;
	const auth = createAuthApi<TIdentity>(client, {
		...options.auth,
		persistToken: false,
	});
	let scope = options.scope ?? null;
	let generation = 0;
	let disposed = false;
	let work: AbortController | null = null;
	const anonymous = (
		status: SessionState<TIdentity>["status"] = "anonymous",
		error: unknown = null,
	): SessionState<TIdentity> => ({
		identity: null,
		namespace: null,
		status,
		error,
	});
	const abortError = () =>
		new DOMException("Session operation superseded", "AbortError");
	const invalidate = () => {
		generation++;
		work?.abort();
		work = null;
	};
	const begin = (signal?: AbortSignal | null) => {
		if (disposed) throw new Error("Session controller disposed");
		invalidate();
		work = new AbortController();
		return {
			generation,
			controller: work,
			signal: signal ? AbortSignal.any([work.signal, signal]) : work.signal,
			version: client.authVersion,
		};
	};
	const owned = (operation: ReturnType<typeof begin>) =>
		!disposed &&
		operation.generation === generation &&
		operation.version === client.authVersion;
	const active = (operation: ReturnType<typeof begin>) =>
		owned(operation) && !operation.signal.aborted;
	const guard = (operation: ReturnType<typeof begin>) => {
		if (!active(operation)) throw abortError();
	};
	const cleanup = async (
		previous: SessionState<TIdentity>,
		next: SessionState<TIdentity>,
		operation: ReturnType<typeof begin>,
	) => {
		if (previous.namespace !== next.namespace)
			await options.beforeSessionChange?.({
				previous,
				next,
				signal: operation.signal,
			});
		guard(operation);
	};
	const commit = async (
		identity: TIdentity,
		operation: ReturnType<typeof begin>,
		token?: string,
	) => {
		guard(operation);
		const previous = store.getSnapshot();
		const next: SessionState<TIdentity> = {
			identity,
			namespace:
				options.getNamespace?.(identity, scope) ??
				previous.namespace ??
				`session:${generation}`,
			status: "authenticated",
			error: null,
		};
		await cleanup(previous, next, operation);
		if (token !== undefined) client.setAuthToken(token);
		store.setState(next);
		work = null;
	};
	const fail = async (error: unknown, operation: ReturnType<typeof begin>) => {
		if (!owned(operation)) return;
		const previous = store.getSnapshot();
		const next = anonymous("error", error);
		store.setState(next);
		try {
			await cleanup(previous, next, operation);
		} catch (cleanupError) {
			if (
				!(
					cleanupError instanceof DOMException &&
					cleanupError.name === "AbortError"
				)
			)
				options.onLifecycleError?.(cleanupError);
		} finally {
			if (owned(operation)) work = null;
		}
	};
	const clear = async () => {
		const previous = store.getSnapshot();
		const operation = begin();
		client.setAuthToken(null);
		operation.version = client.authVersion;
		const next = anonymous();
		store.setState(next);
		await cleanup(previous, next, operation);
		work = null;
	};
	const unsubscribe = client.onUnauthorized((_error, context) => {
		// Transport captures the affected generation before invoking any observers.
		if (disposed || context.authVersion !== client.authVersion) return;
		void clear().catch((error) => {
			if (error instanceof DOMException && error.name === "AbortError") return;
			options.onLifecycleError?.(error);
		});
	});
	return {
		async login(
			credentials: LoginCredentials,
			input: AuthRequestOptions<LoginResponse<TIdentity>> = {},
		) {
			const previous = store.getSnapshot();
			const operation = begin(input.signal);
			client.setAuthToken(null);
			operation.version = client.authVersion;
			const pending = anonymous("authenticating");
			store.setState(pending);
			try {
				await cleanup(previous, pending, operation);
				const result = await auth.login(credentials, {
					...input,
					signal: operation.signal,
				});
				await commit(result.userData, operation, result.token);
				return result.userData;
			} catch (error) {
				await fail(error, operation);
				throw error;
			}
		},
		async restore(input: AuthRequestOptions<TIdentity> = {}) {
			const operation = begin(input.signal);
			if (!client.getAuthToken()) {
				await clear();
				return null;
			}
			store.setState({
				...store.getSnapshot(),
				status: "restoring",
				error: null,
			});
			try {
				const identity = await auth.me({ ...input, signal: operation.signal });
				await commit(identity, operation);
				return identity;
			} catch (error) {
				await fail(error, operation);
				throw error;
			}
		},
		async logout(input: AuthRequestOptions<void> = {}) {
			const previous = store.getSnapshot();
			const operation = begin(input.signal);
			// Auth API captures the old token and clears it synchronously, before its first await.
			const remote = auth.logout({ ...input, signal: operation.signal });
			operation.version = client.authVersion;
			const next = anonymous();
			store.setState(next);
			// Settle both so a failing server or cache hook never creates an unhandled rejection.
			const results = await Promise.allSettled([
				remote,
				cleanup(previous, next, operation),
			]);
			if (active(operation)) work = null;
			for (const result of results)
				if (result.status === "rejected") throw result.reason;
		},
		async setScope(nextScope: string | null) {
			if (!options.getNamespace)
				throw new Error("setScope requires getNamespace");
			if (nextScope === scope) return;
			const previous = store.getSnapshot();
			const operation = begin();
			scope = nextScope;
			if (previous.identity === null) {
				work = null;
				store.setState(anonymous());
				return;
			}
			store.setState({ ...previous, status: "restoring", error: null });
			try {
				await commit(previous.identity, operation);
			} catch (error) {
				await fail(error, operation);
				throw error;
			}
		},
		dispose() {
			if (disposed) return;
			disposed = true;
			invalidate();
			unsubscribe();
		},
	};
}

	input: unknown,

export async function createValidatedEndpoint<T>(
	schema: ResponseValidator<T>,
	fetcher: () => Promise<unknown>,
): Promise<T> {
	const data = await fetcher();

		return await validateResponse(schema, data);

			code: "VALIDATION_ERROR",

export const responseMode: unique symbol = Symbol("sdk-api-response-mode");

/** Fix the typed-view backing-buffer edge, then delegate representations to Axios. */
export const prepareBody: AxiosRequestTransformer = function (
	data: unknown,
	headers,
) {
	if (ArrayBuffer.isView(data)) {
		const view = data as ArrayBufferView;
		return new Uint8Array(view.buffer, view.byteOffset, view.byteLength).slice()
			.buffer;
	}
	if (
		typeof FormData !== "undefined" &&
		data instanceof FormData &&
		!String(headers.getContentType() ?? "").includes("application/json")
	) {
		headers.delete("Content-Type");
	}
	if (data === null || typeof data === "number" || typeof data === "boolean") {
		headers.setContentType("application/json", false);
	}
	if (
		typeof data === "bigint" ||
		typeof data === "function" ||
		typeof data === "symbol"
	) {
		throw configError("Body is not JSON-serializable");
	}
	return data;
};

const nativeResponseTransforms = Array.isArray(axios.defaults.transformResponse)
	? axios.defaults.transformResponse
	: axios.defaults.transformResponse
		? [axios.defaults.transformResponse]
		: [];

/** Native Axios JSON parsing with strict success and lenient rejected error bodies. */
export const parseResponse: AxiosResponseTransformer = function (
	data: unknown,
	headers,
	status,
) {
	if (this.method === "head" || status === 204 || status === 205) {
		return undefined;
	}
	const media = String(headers.get("Content-Type") ?? "")
		.split(";", 1)[0]
		?.trim()
		.toLowerCase();
	const requested =
		(this as InternalAxiosRequestConfig & { [responseMode]?: string })[
			responseMode
		] ?? this.responseType;
	const json =
		requested === "json" ||
		((!requested || requested === "auto") &&
			(media === "application/json" || media?.endsWith("+json")));
	if (!json || typeof data !== "string") return data;
	if (data === "") return undefined;
	const accepted =
		status === undefined || !this.validateStatus || this.validateStatus(status);
	const config: InternalAxiosRequestConfig = {
		...this,
		responseType: "json",
		transitional: {
			...this.transitional,
			silentJSONParsing: !accepted,
			forcedJSONParsing: true,
		},
	};
	let value: unknown = data;
	for (const transform of nativeResponseTransforms) {
		value = transform.call(config, value, headers, status);
	}
	return value;
};

export function isStreamBody(value: unknown): boolean {
	return (
		(typeof ReadableStream !== "undefined" &&
			value instanceof ReadableStream) ||
		(typeof value === "object" &&
			value !== null &&
			"pipe" in value &&
			typeof value.pipe === "function")
	);
}

/** Close unused Node/web stream data without importing Node into browser bundles. */
export function discardData(data: unknown): void {
	if (typeof data !== "object" || data === null) return;
	if ("destroy" in data && typeof data.destroy === "function") data.destroy();
	else if (
		"cancel" in data &&
		typeof data.cancel === "function" &&
		!("locked" in data && data.locked)
	) {
		void Promise.resolve(data.cancel()).catch(() => {});
	}
}

export function nonnegative(value: number, name: string): number {

		throw configError(`Invalid ${name}`);
	}
	return value;

export function createScope(
	parent: AbortSignal | null | undefined,
	owner: AbortSignal,
	timeoutMs: number,
) {
	const controller = new AbortController();
	const cleanup: (() => void)[] = [];
	for (const signal of [parent, owner]) {
		if (!signal) continue;
		const relay = () => controller.abort(signal.reason);
		if (signal.aborted) relay();
		else {
			signal.addEventListener("abort", relay, { once: true });
			cleanup.push(() => signal.removeEventListener("abort", relay));
		}
	}
	const timer =
		timeoutMs > 0
			? setTimeout(
					() =>
						controller.abort(
							new ApiError(0, "Operation timed out", undefined, "", {
								code: "TIMEOUT",
							}),
						),
					timeoutMs,
				)
			: undefined;
	return {
		signal: controller.signal,
		close() {
			if (timer !== undefined) clearTimeout(timer);
			for (const fn of cleanup) fn();
		},
	};
}

export function throwIfAborted(signal: AbortSignal): void {
	if (!signal.aborted) return;
	if (signal.reason instanceof ApiError && signal.reason.code === "TIMEOUT") {
		throw signal.reason;
	}
	throw new ApiError(0, "Operation canceled", undefined, "", {
		code: "ABORTED",
		cause: signal.reason,
	});
}

/** Immediate waiter cancellation, including adapters/hooks that ignore the signal. */
export function waitFor<T>(
	value: PromiseLike<T>,
	signal: AbortSignal,
): Promise<T> {
	return new Promise((resolve, reject) => {
		const abort = () => {
			signal.removeEventListener("abort", abort);
			try {
				throwIfAborted(signal);
			} catch (error) {
				reject(error);
			}
		};
		if (signal.aborted) {
			Promise.resolve(value).catch(() => {});
			abort();
			return;
		}
		signal.addEventListener("abort", abort, { once: true });
		Promise.resolve(value)
			.then(resolve, reject)
			.finally(() => signal.removeEventListener("abort", abort));
	});
}

export function pause(ms: number, signal: AbortSignal): Promise<void> {
	return new Promise((resolve, reject) => {
		const abort = () => {
			clearTimeout(timer);
			signal.removeEventListener("abort", abort);
			try {
				throwIfAborted(signal);
			} catch (error) {
				reject(error);
			}
		};
		const timer = setTimeout(() => {
			signal.removeEventListener("abort", abort);
			resolve();
		}, ms);
		if (signal.aborted) abort();
		else signal.addEventListener("abort", abort, { once: true });
	});
}

export function retryAfter(value: string | null): number | undefined {
	if (value === null) return undefined;
	if (/^\d+$/.test(value.trim())) return Number(value) * 1000;
	// HTTP-date begins with a weekday; Date.parse otherwise accepts unrelated inputs such as "1.5".
	if (
		!/^(?:Mon|Tue|Wed|Thu|Fri|Sat|Sun)(?:,|day,|sday,|nesday,|rsday,|urday,|\s)/i.test(
			value.trim(),
		)
	) {
		return undefined;
	}
	const at = Date.parse(value);
	return Number.isFinite(at) ? Math.max(0, at - Date.now()) : undefined;
}

import { configError } from "./control.ts";

function parseHttp(value: string): URL {
	let url: URL;
	try {
		url = new URL(value);
	} catch (cause) {
		throw configError("Invalid HTTP URL", cause);
	}
	if (!/^https?:$/.test(url.protocol) || url.username || url.password) {
		throw configError("Only HTTP(S) URLs without userinfo are supported");
	}
	return url;
}

function hasInvalidBaseChars(str: string): boolean {
	for (let i = 0; i < str.length; i++) {
		const code = str.charCodeAt(i);
		if (code <= 0x1f || code === 0x7f || code === 0x5c) return true;
	}
	return /\s/.test(str);
}

function hasInvalidPathChars(str: string): boolean {
	for (let i = 0; i < str.length; i++) {
		const code = str.charCodeAt(i);
		if (code <= 0x20 || code === 0x7f || code === 0x5c) return true;
	}
	return false;
}

export function validateBase(base: string): string {
	if (hasInvalidBaseChars(base)) throw configError("Invalid baseUrl");
	if (!base) return "";
	if (base.startsWith("//")) {
		throw configError("Protocol-relative baseUrl is not supported");
	}
	if (base.startsWith("/")) {
		const normalized = new URL(base, "https://sdk.invalid");
		if (normalized.search || normalized.hash) {
			throw configError("baseUrl must not contain a query or fragment");
		}
		return normalized.pathname.replace(/\/+$/, "");
	}
	if (!/^https?:\/\//i.test(base)) {
		throw configError(
			"baseUrl must be absolute HTTP(S) or a root-relative prefix",
		);
	}
	const url = parseHttp(base);
	if (url.search || url.hash) {
		throw configError("baseUrl must not contain a query or fragment");
	}
	return url.href.replace(/\/+$/, "");
}

export function validateOrigins(origins: readonly string[]): Set<string> {
	return new Set(
		origins.map((value) => {
			const url = parseHttp(value);
			if (url.pathname !== "/" || url.search || url.hash) {
				throw configError("allowedOrigins must contain origins only");
			}
			return url.origin;
		}),
	);
}

export function resolveUrl(
	base: string,
	path: string,
	allowed: Set<string>,
): { url: string; trusted: boolean } {
	if (
		typeof path !== "string" ||
		!path ||
		hasInvalidPathChars(path) ||
		path.startsWith("//")
	) {
		throw configError("Invalid request path");
	}
	const absolute = /^https?:\/\//i.test(path);
	if (!absolute && /^[a-z][a-z\d+.-]*:/i.test(path)) {
		throw configError("Unsupported request URL scheme");
	}
	const browserOrigin =
		typeof window !== "undefined" && /^https?:$/.test(window.location?.protocol)
			? window.location.origin
			: undefined;
	const knownOrigin =
		base && /^https?:\/\//i.test(base) ? parseHttp(base).origin : browserOrigin;
	const constructed = absolute ? path : `${base}/${path.replace(/^\/+/, "")}`;
	const parsed = /^https?:\/\//i.test(constructed)
		? parseHttp(constructed)
		: new URL(constructed, browserOrigin ?? "https://sdk.invalid");
	if (
		absolute &&
		knownOrigin &&
		parsed.origin !== knownOrigin &&
		!allowed.has(parsed.origin)
	) {
		throw configError("Request origin is not trusted");
	}
	if (!absolute && base) {
		const prefix = new URL(
			base,
			browserOrigin ?? "https://sdk.invalid",
		).pathname.replace(/\/+$/, "");
		if (
			prefix &&
			parsed.pathname !== prefix &&
			!parsed.pathname.startsWith(`${prefix}/`)
		) {
			throw configError("Request path escapes baseUrl prefix");
		}
	}
	parsed.hash = "";
	const trusted =
		(knownOrigin !== undefined && parsed.origin === knownOrigin) ||
		allowed.has(parsed.origin) ||
		(!absolute && !knownOrigin);
	return {
		url: /^https?:\/\//i.test(constructed)
			? parsed.href
			: `${parsed.pathname}${parsed.search}`,
		trusted,
	};
}

	createApiClient,

/** Custom adapters, like built-ins, own HTTP settlement. Axios dispatch does not re-settle. */
function adapter(
	status: number,
	data: unknown,
	onRequest?: (response: AxiosResponse) => void,
): AxiosAdapter {
	return async (config) => {
		const response: AxiosResponse = {
			status,
			statusText: status < 300 ? "OK" : "Rejected",
			data,
			headers: { "content-type": "application/json", "x-trace": "sdk" },
			config,
		};
		onRequest?.(response);
		if (config.validateStatus && !config.validateStatus(status)) {
			throw new AxiosError(
				"Request failed with status code " + status,
				status >= 500
					? AxiosError.ERR_BAD_RESPONSE
					: AxiosError.ERR_BAD_REQUEST,
				config,
				undefined,
				response,
			);
		}
		return response;
	};
}

import type { Awaitable } from "../src/mod.ts";

export function mockFetch(
	fn: (url: string, init: RequestInit) => Awaitable<Response>,
): typeof fetch {
	return async (input, init) =>
		fn(input instanceof Request ? input.url : String(input), init ?? {});
}

export function json(
	data: unknown,
	status = 200,
	headers: HeadersInit = {},
): Response {
	const merged = new Headers(headers);
	merged.set("Content-Type", "application/json");
	return new Response(JSON.stringify(data), { status, headers: merged });
}

export function deferred<T>() {
	let resolve!: (value: T) => void;
	let reject!: (reason?: unknown) => void;
	const promise = new Promise<T>((yes, no) => {
		resolve = yes;
		reject = no;
	});
	return { promise, resolve, reject };
}

export const sleep = (ms: number) =>
	new Promise<void>((resolve) => setTimeout(resolve, ms));
